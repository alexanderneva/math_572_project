The initial value problem for model 1 is given by

  \begin{bmatrix}
    x^{\prime}\\

  \end{bmatrix} = 
  \begin{bmatrix}
    0 & a \\

    \end{bmatrix}

\end{equation}
where 
\[
  q:=\begin{bmatrix}r\\ s\end{bmatrix},r,s\in\R.

\section{Solving Model 1}
Following \cite[Chap. 7~]{boyce2013elementary} to solve \ref{model1} we look for a fundamental solution set given by the eigenvectors of the matrix 
\begin{equation}
  A := \begin{bmatrix}
    0 & a \\
    b & 0 \\
  \end{bmatrix}.
\end{equation}
The characteristic polynomial of $M$ is 
\begin{eqnarray}
  p(\lambda) &= \lambda^2 - \text{Tr} A \lambda + \det{A}\\
	     &= \lambda^2 - ab
\end{eqnarray}
Setting $p=0$ and solving for $\lambda$ yields
\begin{align}
  \lambda_+ = \sqrt{ab},\quad \lambda_- = -\sqrt{ab}.
\end{align}
These will be real valued when $\text{sign}(a)=\text{sign}(b)$ and a $\C$-conjugate pair otherwise. The eigenvector $e_+$ comes from
\begin{align*}
  M-\lambda_+ I &= 
    \begin{bmatrix}
      -\sqrt{ab} & a \\
      b  & -\sqrt{ab}
    \end{bmatrix} & \\
   &= \begin{bmatrix}-\sqrt{ab}&a\\0&0\end{bmatrix} & \left(\frac{b}{\sqrt{ab}}R_1 \to R_2\right)\\
\end{align*}
Hence, $v_1 = \left(\sqrt{\frac{a}{b}},1\right)$ is an eigenvector for matrix $M$. Similarly, we can find $v_2 = \left(1,-\sqrt{\frac{b}{a}}\right)$. We normalize $v_1$ and $v_2$ before defining our transformation matrix.
\begin{align*}
  \xi_1:=v_1/||v_1||=\sqrt{\frac{a+b}{b}}^{-1}\left(\sqrt{\frac{a}{b}},1\right)=\left(\sqrt{\frac{a}{a+b}},\sqrt{\frac{b}{a+b}}\right)\\
  \xi_2:=v_2/||v_2||=\sqrt{\frac{a+b}{a}}^{-1}\left(\sqrt{1,-\frac{b}{a}}\right)=\left(\sqrt{\frac{a}{a+b}},-\sqrt{\frac{b}{a+b}}\right)\\
\end{align*}
Let $P$ denote the transformation matrix
\begin{equation*}
  P:=\begin{bmatrix}x_1\\x_2\end{bmatrix}=\begin{bmatrix}\sqrt{\frac{a}{a+b}} & \sqrt{\frac{a}{a+b}}\\
  \sqrt{\frac{b}{a+b}} & -\sqrt{\frac{b}{a+b}}
\end{bmatrix}
\end{equation*}
$P$ is singular when 

\begin{equation*}
  0=\begin{vmatrix}\sqrt{\frac{a}{a+b}} & \sqrt{\frac{a}{a+b}}\\
    \sqrt{\frac{b}{a+b}} & -\sqrt{\frac{b}{a+b}}
  \end{vmatrix} =\frac{-2\sqrt{ab}}{a+b}
\end{equation*}
So long as $a\neq0$ and $b\neq 0$, $P$ is invertible. We are assuming that $a,b>0$, so the case $a+b\neq0$ does not apply. $P^{-1}$ is given by
\[
  \frac{a+b}{-2\sqrt{ab}}
 \begin{bmatrix}
    -\sqrt{\frac{b}{a+b}} & -\sqrt{\frac{a}{a+b}}\\
    -\sqrt{\frac{b}{a+b}} & \sqrt{\frac{a}{a+b}}
  \end{bmatrix}
  =\frac{1}{2}\begin{bmatrix}
    \sqrt{\frac{a+b}{a}} & \sqrt{\frac{a+b}{b}}\\
    \sqrt{\frac{a+b}{a}} & -\sqrt{\frac{a+b}{b}}
  \end{bmatrix}
\]
We can change coordinates
\begin{equation*}
  \begin{bmatrix}u\\v\end{bmatrix}=P^{-1}\begin{bmatrix}x\\y\end{bmatrix}.
\end{equation*}
For the time derivative, this also implies
\begin{align*}
  \begin{bmatrix}
    u^\prime\\
    v^\prime
  \end{bmatrix}&=
  P^{-1}\begin{bmatrix}
    x^\prime\\y^\prime
  \end{bmatrix}\\
  &= P^{-1}AP\begin{bmatrix}
    u\\v 
  \end{bmatrix}\\
  &= D\begin{bmatrix}
    u\\v
  \end{bmatrix}\\
\end{align*}
where $D$ is given by
\begin{equation*}
  D:= \begin{bmatrix}\sqrt{ab} & 0 \\ 0 & -\sqrt{ab}\end{bmatrix}.
\end{equation*}
The initial conditions become 
\[
  \begin{bmatrix}u_0\\v_0\end{bmatrix} = P^{-1}\begin{bmatrix}x_0 \\ y_0 \end{bmatrix}
\]
from which we can recover
\begin{equation}\label{initial1}
  \begin{bmatrix}
    u_0\\v_0 
  \end{bmatrix} 
  =\frac{1}{2}\begin{bmatrix}
    x_0\sqrt{\frac{a+b}{a}} +y_0\sqrt{\frac{a+b}{b}}\\
    x_0\sqrt{\frac{a+b}{a}} -y_0\sqrt{\frac{a+b}{b}}
  \end{bmatrix}
\end{equation}
Each coordinate can be solved independently yielding
\begin{align*}
  \psi(t)=\begin{bmatrix}\psi_1(t)\\\psi_2(t)\end{bmatrix}&=\begin{bmatrix}u_0e^{\sqrt{ab}t}\\v_0e^{-\sqrt{ab}t}\end{bmatrix}\\.
\end{align*}
We can bring $\psi(t)$ back to the $(x,y)$ coordinates by $\phi(t):=P\psi(t)$
\begin{align}\label{general}
  \phi(t)=P\psi(t)&= \begin{bmatrix}
  \sqrt{\frac{a}{a+b}} & \sqrt{\frac{a}{a+b}}\\
\sqrt{\frac{b}{a+b}} & -\sqrt{\frac{b}{a+b}}\end{bmatrix}\begin{bmatrix}u_0e^{\sqrt{ab}t}\\v_0e^{-\sqrt{ab}t}\end{bmatrix}\\
		     &= u_0e^{\sqrt{ab}t}\xi_1+v_0e^{-\sqrt{ab}t}\xi_2
\end{align}
The Wronskian of these vectors is 
\begin{align*}

      &= \begin{vmatrix}
	e^{\sqrt{ab}t}\sqrt{\frac{a}{a+b}} & e^{-\sqrt{ab}t}\sqrt{\frac{a}{a+b}}\\
	e^{\sqrt{ab}t}\sqrt{\frac{b}{a+b}} & -e^{-\sqrt{ab}t}\sqrt{\frac{b}{a+b}}
      \end{vmatrix}\\
      & = \frac{-2\sqrt{ab}}{a+b}\neq 0\quad\text{If }a>0,b>0.
\end{align*}A general solution for the IVP \ref{model1} is

\subsection{To show the solutions are hyperbolic}
Following the procedure in \cite[Chap. 9~]{boyce2013elementary} we can view our system as 
\begin{equation}
  \begin{bmatrix}
    x^\prime\\
  y^\prime	
  \end{bmatrix}=\begin{bmatrix}
  F(x,y)\\
  G(x,y)
\end{bmatrix},\quad F(x,y)=ay,~G(x,y)=bx,~ a,b\in\R_+
\end{equation}
with $x(0)=x_0$ and $y(0)=y_0$. We can reduce the order of the system by division
\begin{align*}
  \frac{dy}{dx}&=\frac{\frac{dy}{dt}}{\frac{dx}{dt}} = \frac{G(x,y)}{F(x,y)}\\
						    &=\frac{bx}{ay}\\
  \iff& ay~dy = bx~dx\\
  \iff& \frac{ay^2}{2} = \frac{bx^2}{2}+C\\
  \iff& ay^2-bx^2 = 2C
\end{align*}
which is an equation of a hyperbola. Substituting in the initial condition gives 
\begin{equation*}
  ay^2-bx^2 = ay_0^2-bx_0^2 = \begin{bmatrix} x & y\end{bmatrix}B\begin{bmatrix} x \\ y\end{bmatrix}
\end{equation*}
where 
\[
  B :=\begin{bmatrix}-b & 0\\
  0 & a\end{bmatrix}
\]
It remains for us to check that $\phi(t)$ satisfies the quadratic form.
\begin{align*}
  \phi(t)^TB\phi(t)&= 
    (u_0e^{\sqrt{ab}t}\xi_1+v_0e^{-\sqrt{ab}t}\xi_2)^TB
    (u_0e^{\sqrt{ab}t}\xi_1+v_0e^{-\sqrt{ab}t}\xi_2)\\
		   &= u_0^2e^{2\sqrt{ab}t}\xi_1^T B\xi_1+v_0^2e^{2\sqrt{ab}t}\xi_2^TB\xi_2 + u_0v_0(\xi_1^T B \xi_2 + \xi_2^T B \xi_1)\\\\
  \xi_1^T B\xi_1 &= \begin{bmatrix}\sqrt{\frac{a}{a+b}}&\sqrt{\frac{b}{a+b}}\end{bmatrix}^T\begin{bmatrix}-b & 0\\
    0 & a\end{bmatrix}\begin{bmatrix}\sqrt{\frac{a}{a+b}} & \sqrt{\frac{b}{a+b}}\end{bmatrix}\\
      &= \begin{bmatrix}\sqrt{\frac{a}{a+b}} & \sqrt{\frac{b}{a+b}}\end{bmatrix}^T\begin{bmatrix}-b\sqrt{\frac{a}{a+b}} & a\sqrt{\frac{b}{a+b}}\end{bmatrix}\\
      &=0\\
\xi_2^T B\xi_2 &= \begin{bmatrix}\sqrt{\frac{a}{a+b}}&-\sqrt{\frac{b}{a+b}}\end{bmatrix}^T\begin{bmatrix}-b & 0\\
    0 & a\end{bmatrix}\begin{bmatrix}\sqrt{\frac{a}{a+b}} & -\sqrt{\frac{b}{a+b}}\end{bmatrix}\\
      &=0\\
    \xi_2^TB\xi_1&=\begin{bmatrix}\sqrt{\frac{a}{a+b}}&-\sqrt{\frac{b}{a+b}}\end{bmatrix}^T\begin{bmatrix}-b & 0\\
    0 & a\end{bmatrix}\begin{bmatrix}\sqrt{\frac{a}{a+b}}&\sqrt{\frac{b}{a+b}}\end{bmatrix}\\
      &= \frac{-2ab}{a+b}\\
    \xi_1^TB\xi_2&=\begin{bmatrix}\sqrt{\frac{a}{a+b}}&\sqrt{\frac{b}{a+b}}\end{bmatrix}^T\begin{bmatrix}-b & 0\\0 & a\end{bmatrix}\begin{bmatrix}\sqrt{\frac{a}{a+b}}&-\sqrt{\frac{b}{a+b}}\end{bmatrix}\\ 
&= \frac{-2ab}{a+b}\\
\implies
    \phi(t)^TB\phi(t)&= \frac{-4u_0v_0ab}{a+b}.
\end{align*}
from \autoref{initial1}, we can calculate
\begin{align*}
  \phi(t)^TB\phi(t)&= \frac{-4u_0v_0ab}{a+b} \\
		   &= 
		   \frac{-4ab}{a+b}\frac{1}{4}\left(x_0\sqrt{\frac{a+b}{a}} +y_0\sqrt{\frac{a+b}{b}}\right)
		 \left(x_0\sqrt{\frac{a+b}{a}} -y_0\sqrt{\frac{a+b}{b}}\right)\\
		   &= \frac{-ab}{a+b}\left(\frac{a+b}{a}x_0^2-\frac{a+b}{b}y_0^2\right)\\
		   &= ay_0^2-bx_0^2
\end{align*}

\section{Solving the Richardson Model}
\subsection{Equilibrium, Nullclines and Stability}
From the Richardson model, \autoref{richardson}, we have the matrix
\begin{equation*}
  R:=\begin{bmatrix}
    -m & a\\
    b & -n\\
  \end{bmatrix}
\end{equation*}
If $\det{R}\neq0,$ we have 
\[
  R^{-1}=\frac{1}{ab-mb}\begin{bmatrix}n & a\\ b & m\end{bmatrix}
\]We look for solution $(x^\star,y^\star)$ satisfying
\begin{align}\label{qstar}
  \begin{bmatrix}0 \\ 0\end{bmatrix}&=\begin{bmatrix}
    -m & a\\
    b & -n\\
    \end{bmatrix}\begin{bmatrix}x^\star\\y^\star\end{bmatrix}+\begin{bmatrix}r\\s\end{bmatrix} \\
    \iff \begin{bmatrix}x^\star \\ y^\star\end{bmatrix}&= -R^{-1}\begin{bmatrix}r\\s\end{bmatrix}\\
					   &=\frac{1}{mn-ab}\begin{bmatrix}nr+as \\ br+ms\end{bmatrix}
\end{align}
If we assume $R$ is invertible, which will be the case so long as we still let $a,b>0$ and $mn\neq ab$, we have the equilibrium point
\begin{equation*}
  \left(x^\star,y^\star\right)=\left(\frac{nr+as}{mn-ab},\frac{br+ms}{mn-ab}\right)
\end{equation*}
Consider the coordinate change
\begin{equation*}
  \begin{bmatrix}x \\ y\end{bmatrix}= \begin{bmatrix}u+x^\star \\ v + y^\star\end{bmatrix}
\end{equation*}
In the new coordinates, we have
\begin{align*}
  \begin{bmatrix}u^\prime\\v^\prime\end{bmatrix}=\begin{bmatrix}x^\prime\\y^\prime\end{bmatrix}&=R\begin{bmatrix}x\\y\end{bmatrix}+\begin{bmatrix}r\\s\end{bmatrix}\\&=R\begin{bmatrix}u+x^\star \\ v + y^\star\end{bmatrix}+\begin{bmatrix}r\\s\end{bmatrix}\\
											       &=R\begin{bmatrix}u\\ v\end{bmatrix}-RR^{-1}\begin{bmatrix}r\\s\end{bmatrix}+\begin{bmatrix}r\\s\end{bmatrix}&\text{\autoref{qstar}}\\
											       &=R\begin{bmatrix}u\\v\end{bmatrix}
\end{align*}
Thus, we can look at the transformed system in $(u,v)$ coordinates
\begin{equation}\label{uvrichardson}
\begin{bmatrix}u^\prime\\v^\prime\end{bmatrix}=R\begin{bmatrix}u\\v\end{bmatrix},\quad u_0=x_0-x^\star,v_0=y_0-y^\star
\end{equation}
We can solve it as we did the mutual fear model \autoref{model1}. We employ the Routh Hurwitz criteria from \\ \cite[Chapter 3.]{MoghadasJaberiDouraki2019} which says that \autoref{richardson} is a stable system iff $\text{Tr }R<0$ and $\det{R}>0$. We have
\begin{align*}
  \text{Tr }J = -(m+n)<0\quad\forall m,n>0\\
  \det{J} = mn-ab > 0\iff mn > ab
\end{align*}
